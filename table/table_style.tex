\newcommand{\facttablecompact}{%
  \setlength{\tabcolsep}{4pt}%
  \renewcommand{\arraystretch}{1.08}%
  \setlength{\aboverulesep}{0.35ex}%
  \setlength{\belowrulesep}{0.35ex}%
  \setlength{\cmidrulekern}{0.4em}%
}
\newenvironment{facttable}{\begingroup\facttablecompact}{\endgroup}
\newcommand{\facttableNumCol}[1]{>{\raggedleft\arraybackslash}p{#1}}
\newcommand{\facttableTextCol}[1]{>{\raggedright\arraybackslash}p{#1}}
\newcolumntype{F}{>{\raggedright\arraybackslash}X}
\newcolumntype{N}{>{\raggedleft\arraybackslash}X}
\newcolumntype{C}{>{\centering\arraybackslash}X}
\newcommand{\facttableTextX}{F}
\newcommand{\facttableNumX}{N}
\newcommand{\facttableCenterX}{C}
\newcommand{\facttableHeaderCell}[1]{\cellcolor{headerwash}\textbf{#1}}
\newcommand{\facttableBestCell}[1]{%
  \begingroup\setlength{\fboxsep}{1.05pt}\setlength{\fboxrule}{0.35pt}%
  \fcolorbox{facttableaccent}{facttablewash}{\bfseries #1}\endgroup}
\newcommand{\facttableBestText}[1]{\textcolor{factblue}{\textbf{#1}}}

\newcommand{\facttableFactRow}{\rowcolor{facttablewash}}
\newcommand{\facttableSummaryCell}[1]{\cellcolor{summarywash}#1}
\newcommand{\facttableFactCell}[1]{\cellcolor{factsummary}#1}
\newcommand{\facttableMetricLabel}[1]{\textcolor{timegray}{#1}}
\newcommand{\facttableMetricBlock}[2]{%
  \begin{tabular}[t]{@{}l@{}}#1\\[-0.25ex]\facttableMetricLabel{#2}\end{tabular}}
\newcommand{\facttableBarCell}[3]{%
